        let currentTabId = 1;

        // Initialize App
        document.addEventListener('DOMContentLoaded', () => {
            renderTabContent(currentTabId);
            populateModalOptions();
        });

        // Switch Tabs Function
        function switchTab(tabId) {
            currentTabId = tabId;
            
            // Clear search query if active
            const searchInput = document.getElementById('searchInput');
            if (searchInput.value) {
                searchInput.value = '';
                document.getElementById('clearSearchBtn').classList.add('hidden');
            }

            renderTabContent(tabId);

            // Smooth scroll to top of services grid if scrolled down
            const gridTop = document.getElementById('sectionHeader').getBoundingClientRect().top + window.scrollY - 100;
            if (window.scrollY > 300) {
                window.scrollTo({ top: gridTop, behavior: 'smooth' });
            }
        }

        // Render Active Tab Content
        function renderTabContent(tabId) {
            const currentData = salonData.find(item => item.id === tabId);
            if (!currentData) return;

            // Update Active UI States for Buttons
            updateTabButtonsState(tabId);

            // Header info
            document.getElementById('sectionTitle').innerText = currentData.title;
            document.getElementById('sectionDesc').innerText = currentData.description;
            document.getElementById('serviceCount').innerText = `${currentData.services.length} услуг(-и)`;

            // Render Cards
            renderCards(currentData.services);
        }

        // Render Cards List
        function renderCards(servicesList) {
            const grid = document.getElementById('servicesGrid');
            const noResults = document.getElementById('noResults');

            if (servicesList.length === 0) {
                grid.innerHTML = '';
                noResults.classList.remove('hidden');
                return;
            }

            noResults.classList.add('hidden');

            grid.innerHTML = servicesList.map(service => `
                <div class="bg-white rounded-2xl p-6 border border-nude-200/80 shadow-sm hover:shadow-xl transition-all duration-300 flex flex-col justify-between group hover:-translate-y-1">
                    <div>
                        <div class="flex justify-between items-start gap-2 mb-3">
                            <h3 class="font-serif-title font-semibold text-lg text-darkCharcoal group-hover:text-nude-500 transition-colors">
                                ${service.name}
                            </h3>
                            ${service.badge ? `
                                <span class="px-2.5 py-0.5 rounded-full text-[10px] font-semibold tracking-wider uppercase bg-nude-100 text-nude-500 border border-nude-200 shrink-0">
                                    ${service.badge}
                                </span>
                            ` : ''}
                        </div>
                        <p class="text-xs text-gray-500 leading-relaxed font-light mb-4">
                            ${service.desc}
                        </p>
                    </div>

                    <div class="pt-4 border-t border-nude-100 flex items-center justify-between mt-auto">
                        <div>
                            <div class="text-base font-bold text-darkCharcoal">
                                ${service.priceRub.toLocaleString()} ₽
                            </div>
                            <div class="text-[11px] text-gray-400 font-medium">
                                ~ ${service.priceKzt.toLocaleString()} ₸
                            </div>
                        </div>

                        <div class="flex items-center gap-3">
                            <span class="text-[11px] text-gray-400 flex items-center gap-1">
                                <i class="fa-regular fa-clock text-nude-400"></i> ${service.duration}
                            </span>
                            <button onclick="openBookingModal('${service.name}')" class="w-9 h-9 rounded-full bg-nude-100 hover:bg-darkCharcoal text-darkCharcoal hover:text-white flex items-center justify-center transition-all duration-300">
                                <i class="fa-solid fa-plus text-xs"></i>
                            </button>
                        </div>
                    </div>
                </div>
            `).join('');
        }

        // Update CSS Classes for Top & Bottom Navigation
        function updateTabButtonsState(activeId) {
            // Update Top Tabs
            for (let i = 1; i <= 5; i++) {
                const topTab = document.getElementById(`top-tab-${i}`);
                if (topTab) {
                    if (i === activeId) {
                        topTab.className = "tab-btn px-5 py-2.5 rounded-full text-xs font-medium uppercase tracking-wider transition-all duration-300 flex items-center gap-2 bg-darkCharcoal text-white border-darkCharcoal shadow-md";
                    } else {
                        topTab.className = "tab-btn px-5 py-2.5 rounded-full text-xs font-medium uppercase tracking-wider transition-all duration-300 flex items-center gap-2 bg-white text-gray-600 border-nude-200 hover:border-nude-400 hover:bg-nude-50";
                    }
                }
            }

            // Update Bottom Fixed Navigation Tabs
            for (let i = 1; i <= 5; i++) {
                const bottomTab = document.getElementById(`bottom-tab-${i}`);
                if (bottomTab) {
                    const iconWrapper = bottomTab.querySelector('div');
                    if (i === activeId) {
                        bottomTab.className = "nav-item flex flex-col items-center gap-1 px-3 py-1 rounded-xl transition-all duration-300 text-roseGold";
                        iconWrapper.className = "w-8 h-8 rounded-full flex items-center justify-center transition-all bg-white/10 shadow-inner";
                    } else {
                        bottomTab.className = "nav-item flex flex-col items-center gap-1 px-3 py-1 rounded-xl transition-all duration-300 text-gray-400 hover:text-white";
                        iconWrapper.className = "w-8 h-8 rounded-full flex items-center justify-center transition-all";
                    }
                }
            }
        }

        // Live Search Handler across all categories
        function handleSearch() {
            const query = document.getElementById('searchInput').value.trim().toLowerCase();
            const clearBtn = document.getElementById('clearSearchBtn');

            if (query.length > 0) {
                clearBtn.classList.remove('hidden');
                
                // Filter all services across all categories
                let allServices = [];
                salonData.forEach(cat => {
                    cat.services.forEach(serv => {
                        if (serv.name.toLowerCase().includes(query) || serv.desc.toLowerCase().includes(query)) {
                            allServices.push(serv);
                        }
                    });
                });

                document.getElementById('sectionTitle').innerText = `Результаты поиска`;
                document.getElementById('sectionDesc').innerText = `По запросу: "${query}"`;
                document.getElementById('serviceCount').innerText = `${allServices.length} найдено`;

                renderCards(allServices);

            } else {
                clearBtn.classList.add('hidden');
                renderTabContent(currentTabId);
            }
        }

        function clearSearch() {
            document.getElementById('searchInput').value = '';
            document.getElementById('clearSearchBtn').classList.add('hidden');
            renderTabContent(currentTabId);
        }

        // Modal Logic
        function populateModalOptions() {
            const select = document.getElementById('modalServiceSelect');
            select.innerHTML = '';
            
            salonData.forEach(cat => {
                const group = document.createElement('optgroup');
                group.label = cat.title;
                cat.services.forEach(serv => {
                    const opt = document.createElement('option');
                    opt.value = serv.name;
                    opt.textContent = `${serv.name} (${serv.priceRub} ₽)`;
                    group.appendChild(opt);
                });
                select.appendChild(group);
            });
        }

        function openBookingModal(preselectedService = null) {
            const modal = document.getElementById('bookingModal');
            const card = document.getElementById('modalCard');
            const select = document.getElementById('modalServiceSelect');

            if (preselectedService) {
                select.value = preselectedService;
            }

            modal.classList.remove('hidden');
            setTimeout(() => {
                modal.classList.remove('opacity-0');
                card.classList.remove('scale-95', 'opacity-0');
                card.classList.add('scale-100', 'opacity-100');
            }, 10);

            document.body.style.overflow = 'hidden';
        }

        function closeBookingModal() {
            const modal = document.getElementById('bookingModal');
            const card = document.getElementById('modalCard');

            card.classList.remove('scale-100', 'opacity-100');
            card.classList.add('scale-95', 'opacity-0');
            
            setTimeout(() => {
                modal.classList.add('hidden');
                document.body.style.overflow = 'auto';
            }, 300);
        }

        function submitBooking(event) {
            event.preventDefault();
            closeBookingModal();
            showToast();
            event.target.reset();
        }

        function showToast() {
            const toast = document.getElementById('toast');
            toast.classList.remove('translate-x-full');
            setTimeout(() => {
                toast.classList.add('translate-x-full');
            }, 4000);
        }

        // Close Modal when clicking background
        document.getElementById('bookingModal').addEventListener('click', (e) => {
            if (e.target.id === 'bookingModal') {
                closeBookingModal();
            }
        });
